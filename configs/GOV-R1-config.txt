!
! Secure Government Office Network Lab
! Device: GOV-R1
! Platform: Cisco 1941
! Sanitized configuration for public GitHub repository
!

hostname GOV-R1

!
! Local administrative account
! Secret intentionally redacted from public repository
!
username secadmin privilege 15 secret <REDACTED>

!
! SSH Management
!
ip domain-name govlab.local
ip ssh version 2

!
! Router-on-a-Stick / Inter-VLAN Routing
!
interface GigabitEthernet0/0
 no ip address
 no shutdown

interface GigabitEthernet0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group ADMIN-RESTRICTIONS in

interface GigabitEthernet0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0

interface GigabitEthernet0/0.50
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0

interface GigabitEthernet0/0.60
 encapsulation dot1Q 60
 ip address 192.168.60.1 255.255.255.0
 ip access-group DMZ-RESTRICTIONS in

!
! Unused physical interface
!
interface GigabitEthernet0/1
 no ip address
 shutdown

!
! Restrict Administration VLAN from Internal Server VLAN
!
ip access-list extended ADMIN-RESTRICTIONS
 deny ip 192.168.10.0 0.0.0.255 192.168.50.0 0.0.0.255
 permit ip any any

!
! Restrict DMZ from initiating access to Internal Servers
!
ip access-list extended DMZ-RESTRICTIONS
 deny ip 192.168.60.0 0.0.0.255 192.168.50.0 0.0.0.255
 permit ip any any

!
! Permit remote router management only from IT/Security VLAN
!
ip access-list standard SSH-MANAGEMENT
 permit 192.168.20.0 0.0.0.255

!
! Secure remote management
! Telnet is not permitted
!
line vty 0 4
 access-class SSH-MANAGEMENT in
 login local
 transport input ssh

!
end
