! =========================================================
! Secure Government Office Network Lab
! Device: WH-R1
! Platform: Cisco ISR4331
! Role: Warehouse Branch Router
! =========================================================

version 16.6.4

hostname WH-R1

ip cef
no ipv6 cef

! ---------------------------------------------------------
! IKE / IPsec VPN
! ---------------------------------------------------------

crypto isakmp policy 10
 encr aes
 authentication pre-share
 group 2

! Pre-shared key intentionally omitted from public repository.
! VPN peer: 10.0.0.1

crypto ipsec transform-set GOV-SET esp-aes esp-sha-hmac

crypto map GOV-VPN 10 ipsec-isakmp
 set peer 10.0.0.1
 set transform-set GOV-SET
 match address VPN-TRAFFIC

! ---------------------------------------------------------
! Warehouse LAN
! ---------------------------------------------------------

interface GigabitEthernet0/0/0
 description WAREHOUSE-LAN
 ip address 192.168.70.1 255.255.255.0
 duplex auto
 speed auto

! ---------------------------------------------------------
! WAN Interface to Head Office
! ---------------------------------------------------------

interface GigabitEthernet0/0/1
 description WAN-LINK-TO-GOV-R1
 ip address 10.0.0.2 255.255.255.252
 duplex auto
 speed auto
 crypto map GOV-VPN

interface GigabitEthernet0/0/2
 no ip address
 duplex auto
 speed auto
 shutdown

interface Vlan1
 no ip address
 shutdown

! ---------------------------------------------------------
! Static Routing
! ---------------------------------------------------------

ip classless

ip route 192.168.10.0 255.255.255.0 10.0.0.1
ip route 192.168.20.0 255.255.255.0 10.0.0.1
ip route 192.168.50.0 255.255.255.0 10.0.0.1
ip route 192.168.60.0 255.255.255.0 10.0.0.1

! ---------------------------------------------------------
! VPN Interesting Traffic
! ---------------------------------------------------------

ip access-list extended VPN-TRAFFIC
 permit ip 192.168.70.0 0.0.0.255 192.168.10.0 0.0.0.255
 permit ip 192.168.70.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip 192.168.70.0 0.0.0.255 192.168.50.0 0.0.0.255
 permit ip 192.168.70.0 0.0.0.255 192.168.60.0 0.0.0.255

! ---------------------------------------------------------
! VTY Lines
! ---------------------------------------------------------

line vty 0 4
 login

end
